// The framework discovers this class automatically (ITraining_Module_Builder):
// "Training/0 Build Everything" builds your module in Order, RegisterModule adds
// it to the shared Training_Modules.asset, and the Bootstrap menu, build settings
// and play-mode redirect all pick it up from there. No shared file needs editing.

using System.Collections.Generic;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using static Training_Builder_Core;

public class M3_Module_Builder : ITraining_Module_Builder{
    private const string Root = "Assets/Members/<YourName>";
    private const string ScenesDir = Root + "/Scenes";
    private const string LessonsDir = Root + "/Training/Lessons";
    private const string LessonPath = LessonsDir + "/M3_Lesson.asset";
    private const string ScenePath = ScenesDir + "/Module3_<Topic>.unity";

    // Build order AND Bootstrap menu position. M1 = 0, M2 = 1 — take the next slot.
    public int Order => 2;

    public void Build(){
        BuildLessonAsset();
        BuildScene();
    }

    // Also reachable on its own from the Training menu. Keep the number unique.
    [MenuItem("Training/3C Build Module3 Scene")]
    public static void BuildMenu(){
        new M3_Module_Builder().Build();
    }

    // ------------------------------------------------------------------
    // 1. Lesson content — the ordered steps the player walks through.
    //    Guided phase runs all steps with highlights; Practice re-runs the
    //    Include_In_Quiz steps without highlights and scores them.
    // ------------------------------------------------------------------

    private void BuildLessonAsset(){
        EnsureFolderPath(LessonsDir);

        Lesson_Definition def = LoadOrCreateLesson(LessonPath);
        def.Module_Id = "M3";                          // unique, stable — used as save key
        def.Scene_Name = "Module3_<Topic>";            // must equal the scene file name
        def.Display_Name = "M3 — <Menu Button Label>";
        def.Quiz_Pass_Threshold = 2;                   // min correct answers to pass practice
        def.Practice_Shuffles_Quiz = true;             // false if the ORDER is what you're teaching
        def.Steps = new List<Lesson_Step>{
            // Info: prose + Continue button.
            Info("intro", "Welcome to <topic>.\n\nPress Continue."),

            // Select_Component: player must click the Component_Marker whose
            // Marker_Id equals the step id (highlighted in guided mode).
            Select("some_part", "Some Part", "what this part does"),
            Select("other_part", "Other Part", "what this one does", false), // false = excluded from quiz

            // Panel_Action: player must press the Startup_Action_Button /
            // Action_Interactable whose Action_Id equals the step id.
            // PanelAction("power_on", "Power button", "Teach why this action matters."),

            // Demo kinds (Axis_Demo, Milling_Demo, or new APPEND-ONLY additions to
            // Lesson_Step_Kind) are handed to your Training_Demo_Controller subclass.
            Info("guided_done", "Guided tour complete — next is the practice quiz.\n\nPress Continue.")
        };
        EditorUtility.SetDirty(def);
    }

    // ------------------------------------------------------------------
    // 2. Scene — everything the module needs, regenerable from this method.
    //    Module scenes are content-only: the XR rig, menu and Lesson_Controller
    //    live in Bootstrap and load your scene additively.
    // ------------------------------------------------------------------

    private void BuildScene(){
        EnsureFolderPath(ScenesDir);
        var scene = NewModuleScene(); // empty scene + light + floor

        // --- Your machine / props ---
        // GameObject machine = (GameObject)PrefabUtility.InstantiatePrefab(
        //     AssetDatabase.LoadAssetAtPath<GameObject>("Assets/<path to your prefab>.prefab"));

        // --- Clickable component markers (for Select_Component steps) ---
        GameObject markerGroup = new GameObject("Component_Markers");
        // BuildComponentMarker sizes the framework marker prefab to world bounds:
        // BuildComponentMarker(markerGroup.transform, "some_part", "Some Part",
        //     RendererBounds(FindChild(machine.transform, "<NodeName>")));

        // --- Prompt panel + results panel + wired Lesson_Sequencer ---
        Module_Scaffold scaffold = BuildLessonScaffold(
            new Vector3(-1.1f, 1.45f, 1.2f),  // prompt panel position
            Quaternion.identity,               // prompt panel rotation
            new Vector3(0f, 1.5f, 1.0f));      // results panel position
        Lesson_Sequencer sequencer = scaffold.Sequencer;

        // Marker registry: wires every Component_Marker under markerGroup to the sequencer.
        Marker_Registry registry = scaffold.Manager.AddComponent<Marker_Registry>();
        SetRef(registry, "MarkersRoot", markerGroup.transform);
        SetRef(registry, "Sequencer", sequencer);
        SetRef(sequencer, "Registry", registry);

        // Guided-mode glow on the current target:
        Part_Highlighter highlighter = scaffold.Manager.AddComponent<Part_Highlighter>();
        SetRef(highlighter, "Sequencer", sequencer);
        SetRef(highlighter, "Registry", registry);

        // For Panel_Action steps, add instead/additionally:
        //   Action_Button_Registry (scans a root for Startup_Action_Button /
        //   Action_Interactable) — build buttons with BuildActionButton(...) and
        //   clickable machine parts with BuildPartAction(...).
        // For demo steps, add your Training_Demo_Controller subclass and
        //   SetRef(sequencer, "DemoController", yourController).

        EditorSceneManager.SaveScene(scene, ScenePath);

        // Adds/updates this module in Assets/Training/Training_Modules.asset —
        // Bootstrap menu, build settings and the play redirect all read it.
        RegisterModule(AssetDatabase.LoadAssetAtPath<Lesson_Definition>(LessonPath), ScenePath);
        Debug.Log("M3_Module_Builder: scene built.");
    }
}
